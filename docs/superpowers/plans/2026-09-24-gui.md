```python
"""Optional progress reporting and cancellation for long engine operations.

Every engine function that accepts ``progress`` and ``cancel`` works exactly
as before when they are omitted. ``cancel`` is only checked between items, so
whatever was written before a cancellation stays valid.
"""

from __future__ import annotations

import threading
from collections.abc import Callable
from dataclasses import dataclass


@dataclass(frozen=True)
class Progress:
    stage: str  # "scan", "files", "notes", "export"
    done: int
    total: int | None
    item: str = ""


ProgressFn = Callable[[Progress], None]


class OperationCancelled(Exception):
    """Raised when the user cancelled; work done so far stays valid."""


class CancelToken:
    """Thread-safe cancellation flag, set from the GUI thread."""

    def __init__(self) -> None:
        self._event = threading.Event()

    def cancel(self) -> None:
        self._event.set()

    @property
    def cancelled(self) -> bool:
        return self._event.is_set()

    def check(self) -> None:
        if self._event.is_set():
            raise OperationCancelled("operation cancelled")


def emit(progress: ProgressFn | None, stage: str, done: int, total: int | None, item: str = "") -> None:
    if progress is not None:
        progress(Progress(stage, done, total, item))


def check_cancel(cancel: CancelToken | None) -> None:
    if cancel is not None:
        cancel.check()
```

```python
def scan_category(
    backup_root: Path,
    category: str,
    key: bytes,
    out_dir: Path,
    *,
    progress: ProgressFn | None = None,
    cancel: CancelToken | None = None,
    done: int = 0,
    total: int | None = None,
) -> CategoryResult:
    category_dir = backup_root / category
    result = CategoryResult(category=category)
    if not category_dir.is_dir():
        return result
    category_out_dir = out_dir / category
    for entry in sorted(p for p in category_dir.rglob("*") if p.is_file()):
        check_cancel(cancel)
        rel_name = entry.relative_to(category_dir).as_posix()
        emit(progress, "scan", done, total, f"{category}/{rel_name}")
        try:
            raw = entry.read_bytes()
        except OSError as exc:
            result.files.append(
                FileResult(
                    rel_name,
                    "error",
                    detail=f"failed to read source file: {type(exc).__name__}: {exc}",
                )
            )
            continue
        _process(rel_name, raw, key, category_out_dir, "", result.files, category)
    return result


def scan_backup(
    backup_root: Path,
    out_dir: Path,
    categories: list[str] | None = None,
    *,
    progress: ProgressFn | None = None,
    cancel: CancelToken | None = None,
) -> BackupReport:
    session_dummy = get_session_dummy(backup_root)
    key = derive_key(session_dummy)
    report = BackupReport()
    selected = list_categories(backup_root)
    if categories is not None:
        wanted = set(categories)
        selected = [c for c in selected if c in wanted]
    total = len(selected)
    for index, category in enumerate(selected):
        check_cancel(cancel)
        report.categories.append(
            scan_category(
                backup_root, category, key, out_dir,
                progress=progress, cancel=cancel, done=index, total=total,
            )
        )
    emit(progress, "scan", total, total)
    return report


def write_report(report: BackupReport, out_dir: Path) -> Path:
    """Write ``report.json`` (what every exporter reads) into ``out_dir``."""
    out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / "report.json"
    path.write_text(json.dumps(report.to_dict(), indent=2), encoding="utf-8")
    return path
```

```python
"""Find the Smart Switch backups on this PC and read their summary."""

from __future__ import annotations

import xml.etree.ElementTree as ET
from collections.abc import Iterator
from dataclasses import dataclass, field
from datetime import datetime
from pathlib import Path

from ssba.core.errors import UnsupportedBackupError

HISTORY_FILE = "backupHistoryInfo.xml"
# <root>\<model>\<model>_<n>\<model>_<timestamp>\backupHistoryInfo.xml
_MAX_DEPTH = 3


@dataclass(frozen=True)
class BackupInfo:
    path: Path
    device_name: str
    model: str
    saved_at: datetime | None
    size: int | None
    items: dict[str, int] = field(default_factory=dict)  # category folder -> items


def _local(tag: str) -> str:
    return tag.rsplit("}", 1)[-1]  # the file uses the "Kies.Common.Data" namespace


def _child_text(element: ET.Element, name: str) -> str:
    for child in element:
        if _local(child.tag) == name:
            return (child.text or "").strip()
    return ""


def _int(text: str) -> int | None:
    try:
        return int(text)
    except ValueError:
        return None


def _time(text: str) -> datetime | None:
    try:
        return datetime.fromisoformat(text)
    except ValueError:
        return None


def _summary(folder: Path, xml: Path) -> BackupInfo:
    try:
        root = ET.parse(xml).getroot()
    except (OSError, ET.ParseError):
        return BackupInfo(folder, "", folder.name, None, None, {})
    bundle = next((e for e in root.iter() if _local(e.tag) == "BundleInfo"), root)
    items: dict[str, int] = {}
    for element in root.iter():
        if _local(element.tag) != "BatchInformation":
            continue
        top = _child_text(element, "FileName").replace("\\", "/").split("/")[0]
        count = _int(_child_text(element, "ItemsCount"))
        if top and count is not None:
            items[top] = items.get(top, 0) + count
    return BackupInfo(
        path=folder,
        device_name=_child_text(bundle, "UserInputName"),
        model=_child_text(bundle, "ModelName") or folder.name,
        saved_at=_time(_child_text(bundle, "SavedTime")),
        size=_int(_child_text(bundle, "FileSize")),
        items=items,
    )


def read_backup(path: Path) -> BackupInfo:
    xml = path / HISTORY_FILE
    if not xml.is_file():
        raise UnsupportedBackupError(f"{path} is not a Smart Switch backup folder (no {HISTORY_FILE})")
    return _summary(path, xml)


def default_roots() -> list[Path]:
    home = Path.home()
    candidates = [
        home / "Samsung" / "SmartSwitch" / "backup",
        home / "Documents" / "Samsung" / "SmartSwitch" / "backup",
    ]
    return [c for c in candidates if c.is_dir()]


def _folders(folder: Path, depth: int) -> Iterator[Path]:
    yield folder
    if depth == 0:
        return
    try:
        children = sorted(p for p in folder.iterdir() if p.is_dir())
    except OSError:
        return
    for child in children:
        yield from _folders(child, depth - 1)


def find_backups(roots: list[Path] | None = None) -> list[BackupInfo]:
    """Every backup at most three folders below a root, newest first."""
    found: dict[Path, BackupInfo] = {}
    for root in default_roots() if roots is None else roots:
        if not root.is_dir():
            continue
        for folder in _folders(root, _MAX_DEPTH):
            xml = folder / HISTORY_FILE
            if xml.is_file():
                found.setdefault(folder.resolve(), _summary(folder, xml))
    return sorted(
        found.values(),
        key=lambda b: (b.saved_at is None, -b.saved_at.timestamp() if b.saved_at else 0.0, str(b.path)),
    )


def format_size(size: int | None) -> str:
    if size is None:
        return "?"
    value = float(size)
    for unit in ("B", "KB", "MB", "GB"):
        if value < 1024:
            return f"{value:.0f} {unit}" if unit == "B" else f"{value:.1f} {unit}"
        value /= 1024
    return f"{value:.1f} TB"
```

```python
from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path

from ssba.core.parsers.apps import AppsExportError, export_apps
from ssba.core.parsers.blocked import BlockedExportError, export_blocked_numbers

    ORIGIN_CATEGORIES,
    STORAGE_CATEGORIES,
    FilesExportError,
    export_files,

from ssba.core.parsers.notes import NotesExportError, export_notes
from ssba.core.parsers.wifi import WifiExportError, export_wifi
from ssba.core.progress import CancelToken, OperationCancelled, ProgressFn


@dataclass(frozen=True)
class Export:
    key: str
    target: str  # name inside the output folder
    categories: tuple[str, ...]  # scan categories that make it available


EXPORTS: tuple[Export, ...] = (
    Export("contacts", "contacts.vcf", ("CONTACT",)),
    Export("messages", "sms.xml", ("GMMESSAGE",)),
    Export("calls", "calls.xml", ("CALLLOG",)),
    Export("wifi", "wifi.html", ("WIFICONFIG",)),
    Export("blocked-numbers", "blocked-numbers.txt", ("CALLOGSETTING",)),
    Export("browser", "browser", ("SBROWSER",)),
    Export("clipboard", "clipboard", ("CLIPBOARD",)),
    Export("clock", "clock.html", ("ALARM", "WORLDCLOCK")),
    Export("notes", "notes", ("SAMSUNGNOTE",)),
    Export("apps", "apps.html", ("APKFILE",)),
    Export("files", "files", STORAGE_CATEGORIES + ORIGIN_CATEGORIES),
)


@dataclass
class ExportOutcome:
    key: str
    ok: bool
    result: object | None
    error: str | None
    path: Path  # what to open


_RUNNERS = {
    "contacts": lambda scan, out, mode, progress, cancel: export_contacts(scan, out),
    "messages": lambda scan, out, mode, progress, cancel: export_messages(scan, out),
    "calls": lambda scan, out, mode, progress, cancel: export_calls(scan, out),
    "wifi": lambda scan, out, mode, progress, cancel: export_wifi(scan, out),
    "blocked-numbers": lambda scan, out, mode, progress, cancel: export_blocked_numbers(scan, out),
    "browser": lambda scan, out, mode, progress, cancel: export_browser(scan, out),
    "clipboard": lambda scan, out, mode, progress, cancel: export_clipboard(scan, out),
    "clock": lambda scan, out, mode, progress, cancel: export_clock(scan, out),
    "notes": lambda scan, out, mode, progress, cancel: export_notes(
        scan, out, progress=progress, cancel=cancel
    ),
    "apps": lambda scan, out, mode, progress, cancel: export_apps(scan, out),
    "files": lambda scan, out, mode, progress, cancel: export_files(
        scan, out, mode, progress=progress, cancel=cancel
    ),
}

_EXPORT_ERRORS = (
    AppsExportError, BlockedExportError, BrowserExportError, CallExportError,
    ClipboardExportError, ClockExportError, ContactExportError, FilesExportError,
    MessageExportError, NotesExportError, WifiExportError,
)


def report_categories(scan_dir: Path) -> set[str]:
    try:
        data = json.loads((scan_dir / "report.json").read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return set()
    categories = data.get("categories") if isinstance(data, dict) else None
    if not isinstance(categories, list):
        return set()
    return {c["category"] for c in categories if isinstance(c, dict) and isinstance(c.get("category"), str)}


def available_exports(scan_dir: Path) -> list[Export]:
    found = report_categories(scan_dir)
    return [e for e in EXPORTS if found.intersection(e.categories)]


def run_export(
    export: Export,
    scan_dir: Path,
    out_dir: Path,
    *,
    files_mode: str = "copy",
    progress: ProgressFn | None = None,
    cancel: CancelToken | None = None,
) -> ExportOutcome:
    path = out_dir / export.target
    try:
        result = _RUNNERS[export.key](scan_dir, path, files_mode, progress, cancel)
    except OperationCancelled:
        raise
    except _EXPORT_ERRORS as exc:
        return ExportOutcome(export.key, False, None, str(exc), path)
    except Exception as exc:  # a bug in one exporter must not stop the others
        return ExportOutcome(export.key, False, None, f"unexpected: {type(exc).__name__}: {exc}", path)
    return ExportOutcome(export.key, True, result, None, path)
```

```python
"""Run engine work off the GUI thread."""

from __future__ import annotations

import logging
import traceback

from PySide6.QtCore import QThread, Signal

from ssba.core.progress import CancelToken, OperationCancelled

log = logging.getLogger("ssba.gui")


class Worker(QThread):
    """Runs ``job(progress, cancel)`` in a thread; results come back as signals."""

    progressed = Signal(object)  # ssba.core.progress.Progress
    succeeded = Signal(object)
    failed = Signal(str)  # traceback text
    cancelled = Signal()

    def __init__(self, job, parent=None) -> None:
        super().__init__(parent)
        self._job = job
        self.token = CancelToken()

    def run(self) -> None:
        try:
            result = self._job(self.progressed.emit, self.token)
        except OperationCancelled:
            log.info("cancelled by the user")
            self.cancelled.emit()
            return
        except Exception:  # shown to the user and logged, never lost
            details = traceback.format_exc()
            log.error("background job failed:\n%s", details)
            self.failed.emit(details)
            return
        self.succeeded.emit(result)
```

```python
"""One localized line per export for the summary page."""

from __future__ import annotations

from ssba.core.recovery import ExportOutcome
from ssba.gui.i18n import tr


def describe(outcome: ExportOutcome) -> tuple[str, str]:
    """(level, text): level is "ok", "warn" (something to check) or "error"."""
    if not outcome.ok:
        return "error", tr("sum_error", error=outcome.error)
    r = outcome.result
    key = outcome.key
    if key == "files" and r is None:
        return "ok", tr("sum_files_kept")
    warn = 0
    if key == "contacts":
        text, warn = tr("sum_contacts", n=r.exported), r.unparsed
    elif key == "messages":
        text, warn = tr("sum_messages", n=r.exported), r.unsupported
    elif key == "calls":
        text, warn = tr("sum_calls", n=r.exported), r.unsupported
    elif key == "wifi":
        text = tr("sum_wifi", n=r.networks, p=r.with_password)
    elif key == "blocked-numbers":
        text = tr("sum_blocked", n=r.exported)
    elif key == "browser":
        text = tr("sum_browser", b=r.bookmarks, h=r.history)
    elif key == "clipboard":
        text, warn = tr("sum_clipboard", t=r.texts, i=r.images), r.missing_images
    elif key == "clock":
        text = tr("sum_clock", a=r.alarms, t=r.timers, c=r.cities)
    elif key == "notes":
        text, warn = tr("sum_notes", n=r.notes, a=r.attachments), r.handwriting_only + r.with_drawings
    elif key == "apps":
        text = tr("sum_apps", n=r.apps)
    else:  # files
        text, warn = tr("sum_files", n=r.placed, m=r.merged), r.conflicts + r.missing
    if warn:
        return "warn", f"{text} \u2014 {tr('sum_check', n=warn)}"
    return "ok", text
```

```python
    def __init__(self, roots: list[Path] | None = None, parent=None) -> None:
        super().__init__(parent)
        from ssba.gui.pages.backup_page import BackupPage
        from ssba.gui.pages.scan_page import ScanPage

        self.roots = roots
        self.state = WizardState()
        self.log_path: Path | None = None
        self.setWizardStyle(QWizard.ModernStyle)
        self.setOption(QWizard.HaveCustomButton1, True)
        self.customButtonClicked.connect(self._on_custom_button)
        self.setPage(PAGE_BACKUP, BackupPage(self))
        self.setPage(PAGE_SCAN, ScanPage(self))
        self._add_later_pages()
        self.resize(760, 560)
        self.retranslate()

    def _add_later_pages(self) -> None:
        """Pages added by later tasks; missing modules are simply skipped."""
        try:
            from ssba.gui.pages.choose_page import ChoosePage
            from ssba.gui.pages.run_page import RunPage
            from ssba.gui.pages.summary_page import SummaryPage
        except ImportError:
            return
        self.setPage(PAGE_CHOOSE, ChoosePage(self))
        self.setPage(PAGE_RUN, RunPage(self))
        self.setPage(PAGE_SUMMARY, SummaryPage(self))

    def _on_custom_button(self, _which: int) -> None:
        # The only custom button is the language switch.
        set_language("en" if language() == "it" else "it")

        self.setButtonText(QWizard.CustomButton1, tr("lang_button"))
        self.setButtonText(QWizard.BackButton, tr("back"))
        self.setButtonText(QWizard.NextButton, tr("next"))
        self.setButtonText(QWizard.CancelButton, tr("cancel"))
        self.setButtonText(QWizard.FinishButton, tr("finish"))
        for page_id in self.pageIds():
            self.page(page_id).retranslate()

    # Dialogs and desktop actions: small methods so tests can replace them.
    def ask(self, text: str) -> bool:
        return QMessageBox.question(self, tr("app_title"), text) == QMessageBox.Yes

    def warn(self, text: str) -> None:
        QMessageBox.warning(self, tr("app_title"), text)

    def info(self, text: str) -> None:

            handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
            handler._ssba = True
            logger.addHandler(handler)
            self.log_path = path
        logger.info("destination %s", dest)
```

```python
def free_space(path: Path) -> int | None:
    """Free bytes on the disk that will hold ``path`` (its nearest existing folder)."""
    probe = path
    while not probe.exists() and probe.parent != probe:
        probe = probe.parent
    try:
        return shutil.disk_usage(probe).free
    except OSError:
        return None

    day = info.saved_at.strftime("%Y-%m-%d") if info.saved_at else "backup"

            self.scan_note.setText(tr("p1_scan_folder", path=folder))
            if not self._dest_edited:
                self.dest.setText(str(folder.parent))
            self.completeChanged.emit()
            return

            info = read_backup(folder)
        except UnsupportedBackupError:
            self.w.warn(tr("p1_not_backup"))
            return

                return False

                    tr("p1_low_space", free=format_size(free), needed=format_size(needed))
                ):
                    return False
            scan_dir = dest / "scan"
            state.reuse_scan = (scan_dir / "report.json").is_file() and self.w.ask(tr("p1_reuse"))
        else:
            scan_dir = state.existing_scan
            state.reuse_scan = True
        state.dest = dest
        state.scan_dir = scan_dir
        self.w.start_log(dest)
        return True

    def nextId(self) -> int:
        from ssba.gui.wizard import PAGE_CHOOSE, PAGE_SCAN

        return PAGE_CHOOSE if self.w.state.reuse_scan else PAGE_SCAN
```

```python
def _decrypt(backup_path, scan_dir, progress, cancel):
```

```python
"""Page 4: run the ticked exports in the background, one row each."""

from __future__ import annotations

import functools
import logging

from PySide6.QtWidgets import QLabel, QProgressBar, QPushButton, QVBoxLayout, QWizardPage

from ssba.core.progress import Progress
from ssba.core.recovery import EXPORTS, ExportOutcome, run_export
from ssba.gui.i18n import tr
from ssba.gui.worker import Worker

log = logging.getLogger("ssba.gui")


def _run_all(exports, scan_dir, dest, files_mode, outcomes, progress, cancel):
    total = len(exports)
    for index, export in enumerate(exports):
        cancel.check()
        progress(Progress("export", index, total, export.key))
        outcome = run_export(
            export, scan_dir, dest, files_mode=files_mode, progress=progress, cancel=cancel,
        )
        log.info("%s: %s", export.key, "ok" if outcome.ok else outcome.error)
        outcomes.append(outcome)
    progress(Progress("export", total, total, ""))
    return outcomes


class RunPage(QWizardPage):
    def __init__(self, wizard) -> None:
        super().__init__()
        self.w = wizard
        self.worker: Worker | None = None
        self._done = False
        self._exports = []
        self._running = ""
        self.labels: dict[str, QLabel] = {}
        self.rows_layout = QVBoxLayout()
        self.bar = QProgressBar()
        self.status = QLabel()
        self.stop = QPushButton()
        layout = QVBoxLayout(self)
        layout.addLayout(self.rows_layout)
        layout.addWidget(self.bar)
        layout.addWidget(self.status)
        layout.addWidget(self.stop)
        layout.addStretch(1)
        self.stop.clicked.connect(self._stop)

    def retranslate(self) -> None:
        self.setTitle(tr("p4_title"))
        self.setSubTitle(tr("p4_subtitle"))
        self.stop.setText(tr("p4_stop"))
        self._refresh_rows()

    def initializePage(self) -> None:
        state = self.w.state
        state.outcomes = []
        self._done = False
        self._running = ""
        keys = list(state.selected) + (["files"] if state.files_mode in ("copy", "move") else [])
        self._exports = [e for e in EXPORTS if e.key in keys]
        while (child := self.rows_layout.takeAt(0)) is not None:
            if child.widget() is not None:
                child.widget().deleteLater()
        self.labels = {}
        for export in self._exports:
            label = QLabel()
            self.labels[export.key] = label
            self.rows_layout.addWidget(label)
        self._refresh_rows()
        self.status.setText("")
        self.stop.setEnabled(True)
        self.bar.setRange(0, 0)
        job = functools.partial(
            _run_all, self._exports, state.scan_dir, state.dest,
            state.files_mode if state.files_mode in ("copy", "move") else "copy", state.outcomes,
        )
        self.worker = Worker(job, self)
        self.worker.progressed.connect(self._on_progress)
        self.worker.succeeded.connect(lambda _outcomes: self._finish(tr("p4_done")))
        self.worker.failed.connect(self._on_failed)
        self.worker.cancelled.connect(lambda: self._finish(tr("p4_cancelled")))
        self.worker.start()

    def _refresh_rows(self) -> None:
        done = {o.key: o for o in self.w.state.outcomes}
        for export in self._exports:
            name = tr(f"export_{export.key}")
            if export.key in done:
                key = "p4_row_ok" if done[export.key].ok else "p4_row_error"
            elif export.key == self._running:
                key = "p4_row_running"
            else:
                key = "p4_row_waiting"
            self.labels[export.key].setText(tr(key, name=name))

    def _on_progress(self, event) -> None:
        if event.stage == "export":
            self._running = event.item
            self.bar.setRange(0, 0)
            self._refresh_rows()
        elif event.total:
            self.bar.setRange(0, event.total)
            self.bar.setValue(event.done)

    def _finish(self, text: str) -> None:
        state = self.w.state
        if state.files_mode == "keep" and state.files_available:
            state.outcomes.append(ExportOutcome("files", True, None, None, state.scan_dir))
        self._running = ""
        self._refresh_rows()
        self.bar.setRange(0, 1)
        self.bar.setValue(1)
        self.stop.setEnabled(False)
        self.status.setText(text)
        self._done = True
        self.completeChanged.emit()

    def _on_failed(self, details: str) -> None:
        self._finish(tr("p4_done"))
        self.w.show_error(details)

    def _stop(self) -> None:
        if self.worker is not None and self.worker.isRunning() and self.w.ask(tr("p4_confirm_stop")):
            self.worker.token.cancel()

    def cleanupPage(self) -> None:
        if self.worker is not None and self.worker.isRunning():
            self.worker.token.cancel()
            self.worker.wait()

    def isComplete(self) -> bool:
        return self._done
```

```python
from dataclasses import dataclass

from PySide6.QtWidgets import QGridLayout, QHBoxLayout, QLabel, QPushButton, QVBoxLayout, QWizardPage

from ssba.gui.i18n import tr
from ssba.gui.summary import describe

_ICONS = {"ok": "\u2714", "warn": "\u26a0", "error": "\u2716"}


@dataclass
class SummaryRow:
    icon: QLabel
    name: QLabel
    text: QLabel
    open_button: QPushButton


class SummaryPage(QWizardPage):
    def __init__(self, wizard) -> None:
        super().__init__()
        self.w = wizard
        self.rows: list[SummaryRow] = []
        self.grid = QGridLayout()
        self.open_results = QPushButton()
        self.guide = QPushButton()
        buttons = QHBoxLayout()
        buttons.addWidget(self.open_results)
        buttons.addWidget(self.guide)
        buttons.addStretch(1)
        layout = QVBoxLayout(self)
        layout.addLayout(self.grid)
        layout.addLayout(buttons)
        layout.addStretch(1)
        self.open_results.clicked.connect(lambda: self.w.open_path(self.w.state.dest))
        self.guide.clicked.connect(lambda: self.w.info(tr("p5_guide_text")))
        self.setFinalPage(True)

    def initializePage(self) -> None:
        while (child := self.grid.takeAt(0)) is not None:
            if child.widget() is not None:
                child.widget().deleteLater()
        self.rows = []
        for index, outcome in enumerate(self.w.state.outcomes):
            row = SummaryRow(QLabel(), QLabel(), QLabel(), QPushButton())
            row.text.setWordWrap(True)
            row.open_button.setEnabled(outcome.path.exists())
            row.open_button.clicked.connect(lambda _checked=False, p=outcome.path: self.w.open_path(p))
            for column, widget in enumerate((row.icon, row.name, row.text, row.open_button)):
                self.grid.addWidget(widget, index, column)
            self.rows.append(row)
        self.retranslate()

    def retranslate(self) -> None:
        self.setTitle(tr("p5_title"))
        self.setSubTitle(tr("p5_subtitle"))
        self.open_results.setText(tr("p5_open_results"))
        self.guide.setText(tr("p5_guide_button"))
        for row, outcome in zip(self.rows, self.w.state.outcomes):
            level, text = describe(outcome)
            row.icon.setText(_ICONS[level])
            row.name.setText(tr(f"export_{outcome.key}"))
            row.text.setText(text)
            row.open_button.setText(tr("p5_open"))
```

```python
# PyInstaller spec: one folder with the GUI and the command-line program.
# Build with packaging/build.ps1.
from pathlib import Path

HERE = Path(SPECPATH)
SRC = str(HERE.parent / "src")

gui = Analysis([str(HERE / "ssba_gui.py")], pathex=[SRC])
cli = Analysis([str(HERE / "ssba_cli.py")], pathex=[SRC], excludes=["PySide6"])

gui_exe = EXE(
    PYZ(gui.pure), gui.scripts, [],
    exclude_binaries=True, name='SmartSwitch Backup Analyzer', console=False,
)
cli_exe = EXE(
    PYZ(cli.pure), cli.scripts, [],
    exclude_binaries=True, name='ssba', console=True,
)
COLLECT(
    gui_exe, gui.binaries, gui.datas,
    cli_exe, cli.binaries, cli.datas,
    name='SmartSwitch-Backup-Analyzer',
)
```

`packaging/build.ps1`:

```powershell
# Build the Windows package: .\packaging\build.ps1
$ErrorActionPreference = "Stop"
$root = Split-Path -Parent $PSScriptRoot
Set-Location $root
$venv = Join-Path $root "build\venv"
if (Test-Path $venv) { Remove-Item -Recurse -Force $venv }
python -m venv $venv
& "$venv\Scripts\python.exe" -m pip install --upgrade pip
& "$venv\Scripts\python.exe" -m pip install ".[gui,build]"
& "$venv\Scripts\pyinstaller.exe" --noconfirm --distpath dist --workpath build\pyinstaller packaging\ssba.spec
$version = (& "$venv\Scripts\python.exe" -c "import importlib.metadata as m; print(m.version('smartswitch-backup-analyzer'))").Trim()
$zip = Join-Path $root "dist\SmartSwitch-Backup-Analyzer-$version-windows.zip"
if (Test-Path $zip) { Remove-Item $zip }
Compress-Archive -Path (Join-Path $root "dist\SmartSwitch-Backup-Analyzer\*") -DestinationPath $zip
Write-Host "Built $zip"
```
